% Title band: {title}{right-hand label}{note}
\newcommand{\tmheader}[3]{%
  \begingroup\setlength{\fboxsep}{9pt}%
  \noindent\colorbox{tmgreen}{\parbox{\dimexpr\textwidth-2\fboxsep}{%
    {\color{white}\fontsize{17}{20}\selectfont\bfseries #1\hfill
     \fontsize{10.5}{20}\selectfont\mdseries #2}\par\vspace{3pt}%
    {\color{white}\fontsize{7.5}{9.5}\selectfont #3}}}\par
  \endgroup
  \nointerlineskip\noindent{\color{tmgold}\rule{\textwidth}{2.5pt}}\par\vspace{9pt}}

% Section heading with a thin rule.
\newcommand{\tmsection}[1]{%
  \par\vspace{10pt}{\fontsize{11}{13}\selectfont\bfseries\color{tmgreen}#1\par}%
  \nointerlineskip\vspace{3.5pt}{\color{tmrule}\hrule height 0.5pt}\vspace{7pt}}

% Small muted note.
\newcommand{\tmnote}[1]{{\fontsize{7}{8.8}\selectfont\color{tmmuted}#1\par}}

% KPI tile: {accent hex}{title}{value}{lines separated by \\}

    {\color[HTML]{#1}\hrule height 2pt}\vspace{5pt}
    {\fontsize{6.3}{7.8}\selectfont\color{tmmuted}\MakeUppercase{#2}\par}\vspace{2.5pt}
    {\fontsize{15}{16}\selectfont\bfseries #3\par}\vspace{3pt}
    {\fontsize{7.3}{9.6}\selectfont #4\par}
  \end{minipage}}

% Chart with a bold title (may be empty) and a muted subtitle:
% {title}{subtitle}{file}. Setting \tmsubh gives the subtitle a fixed height,
% so charts placed side by side line up.
\newlength{\tmsubh}
\setlength{\tmsubh}{0pt}
\newcommand{\tmchart}[3]{%
  \if\relax\detokenize{#1}\relax\else
    {\raggedright\fontsize{9}{11}\selectfont\bfseries #1\par}\vspace{1pt}%
  \fi
  \ifdim\tmsubh>0pt
    \vbox to \tmsubh{\raggedright\fontsize{6.8}{8.4}\selectfont\color{tmmuted}#2\par\vfil}%
  \else
    {\raggedright\fontsize{6.8}{8.4}\selectfont\color{tmmuted}#2\par}%
  \fi
  \vspace{3pt}\includegraphics[width=\linewidth]{#3}\par}
